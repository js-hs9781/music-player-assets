import { FilesetResolver, HandLandmarker } from "https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.0";


// ==========================================

// 🔧 GitHub 資源設定區

// ==========================================

const CONFIG = {

  coverUrl: "https://raw.githubusercontent.com/js-hs9781/music-player-assets/main/final%20cover.png", 

  musicList: [

    "https://raw.githubusercontent.com/js-hs9781/music-player-assets/main/Music1.mp3",

    "https://raw.githubusercontent.com/js-hs9781/music-player-assets/main/Music2.mp3",

    "https://raw.githubusercontent.com/js-hs9781/music-player-assets/main/Music3.mp3",

    "https://raw.githubusercontent.com/js-hs9781/music-player-assets/main/Music4.mp3",

    "https://raw.githubusercontent.com/js-hs9781/music-player-assets/main/Music5.mp3",

    "https://raw.githubusercontent.com/js-hs9781/music-player-assets/main/Music6.mp3"

  ]

};


// ==========================================

// 核心變數

// ==========================================

let handLandmarker = undefined;

let runningMode = "VIDEO";

let lastVideoTime = -1;

let isPlaying = false;

let currentVolume = 0.5;

let smoothedVolume = 0.5;

let currentSongIndex = 0;

let lastGestureTime = 0;

let pauseProtectionTime = 0; // 防誤觸時間

let audioInitialized = false;

let frameCounter = 0;

const SKIP_FRAMES = 2;


const video = document.getElementById("webcam");

const canvasElement = document.getElementById("output_canvas");

const canvasCtx = canvasElement.getContext("2d");

const audioPlayer = document.getElementById("audioPlayer");

const bgCover = document.getElementById("bg-cover");

const protectionMsg = document.getElementById("protection-msg");


// 設定封面圖

bgCover.src = CONFIG.coverUrl;

bgCover.onerror = function() {

    this.src = "https://images.unsplash.com/photo-1470225620780-dba8ba36b745?w=800&q=80";

};


audioPlayer.addEventListener('error', function(e) {

    console.error("Audio Error:", e);

    isPlaying = false;

    updateStatusUI();

});


// ==========================================

// 1. 初始化 AI 模型

// ==========================================

async function createHandLandmarker() {

  const loadingText = document.getElementById("loading-text");

  const startBtn = document.getElementById("startBtn");

  

  loadingText.style.display = "block";

  startBtn.disabled = true;

  startBtn.innerText = "載入中...";

  

  try {

    const vision = await FilesetResolver.forVisionTasks(

      "https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.0/wasm"

    );

    handLandmarker = await HandLandmarker.createFromOptions(vision, {

      baseOptions: {

        modelAssetPath: `https://storage.googleapis.com/mediapipe-models/hand_landmarker/hand_landmarker/float16/1/hand_landmarker.task`,

        delegate: "GPU"

      },

      runningMode: runningMode,

      numHands: 1

    });

    

    loadingText.style.display = "none";

    startBtn.disabled = false;

    startBtn.innerText = "🔊 點此啟動體驗";

  } catch (e) {

    alert("AI 模型載入失敗: " + e);

  }

}

createHandLandmarker();


// ==========================================

// 2. 音樂播放邏輯

// ==========================================

function playNextSong() {

  if (CONFIG.musicList.length === 0) return;

  currentSongIndex = (currentSongIndex + 1) % CONFIG.musicList.length;

  const url = CONFIG.musicList[currentSongIndex];

  audioPlayer.src = url;

  audioPlayer.volume = currentVolume;

  const playPromise = audioPlayer.play();

  if (playPromise !== undefined) {

    playPromise.then(_ => {

      isPlaying = true;

      updateStatusUI();

    }).catch(error => { console.error("Auto-play prevented:", error); });

  }

  document.getElementById("song-name").innerText = `Track #${currentSongIndex + 1}`;

}


audioPlayer.onended = () => playNextSong();


// ==========================================

// 3. 啟動與權限

// ==========================================

window.startApp = function() {

  if (!handLandmarker) return;

  if (!audioInitialized && CONFIG.musicList.length > 0) {

      audioPlayer.src = CONFIG.musicList[0];

      audioPlayer.volume = 0;

      audioPlayer.play().then(() => {

          audioPlayer.pause();

          audioPlayer.currentTime = 0;

          audioPlayer.volume = currentVolume;

          audioInitialized = true;

          console.log("Audio Unlocked");

      }).catch(e => console.log("Unlock skipped:", e));

  }

  

  // 響應式攝影機設定：使用 ideal 來適應不同裝置

  const constraints = {

      video: {

          facingMode: "user",

          width: { ideal: 640 },

          height: { ideal: 480 }

      }

  };


  navigator.mediaDevices.getUserMedia(constraints).then((stream) => {

    video.srcObject = stream;

    video.addEventListener("loadeddata", predictWebcam);

    document.getElementById("startBtn").style.display = "none"; 

    document.getElementById("loading-text").innerText = "啟動成功！請比出 OK 手勢";

    document.getElementById("loading-text").style.display = "block";

  });

}


// ==========================================

// 4. 偵測主迴圈

// ==========================================

async function predictWebcam() {

  // 確保 Canvas 大小跟隨 Video 實際大小 (響應式關鍵)

  if (video.videoWidth > 0 && canvasElement.width !== video.videoWidth) {

      canvasElement.width = video.videoWidth;

      canvasElement.height = video.videoHeight;

  }

  

  if (frameCounter % (SKIP_FRAMES + 1) === 0) {

      if (lastVideoTime !== video.currentTime) {

        lastVideoTime = video.currentTime;

        const startTimeMs = performance.now();

        const results = handLandmarker.detectForVideo(video, startTimeMs);


        canvasCtx.clearRect(0, 0, canvasElement.width, canvasElement.height);

        

        // 顯示防誤觸提示

        if (Date.now() < pauseProtectionTime) {

            protectionMsg.style.display = "block";

        } else {

            protectionMsg.style.display = "none";

        }

        

        if (results.landmarks) {

          for (const landmarks of results.landmarks) {

            const gestureState = handleGestures(landmarks);

            drawHand(landmarks, gestureState); 

          }

        }

      }

  }

  frameCounter++;

  window.requestAnimationFrame(predictWebcam);

}


// ==========================================

// 5. 手勢判定邏輯 (含防誤觸)

// ==========================================

function handleGestures(lm) {

    const now = Date.now();

    let gestureType = "NORMAL"; 


    const thumbTip = lm[4];

    const indexTip = lm[8];

    const middleTip = lm[12];

    const ringTip = lm[16];

    const pinkyTip = lm[20];

    

    const indexPip = lm[6].y;

    const middlePip = lm[10].y;

    const ringPip = lm[14].y;

    const pinkyPip = lm[18].y;


    const isIndexExt = indexTip.y < indexPip;

    const isMiddleExt = middleTip.y < middlePip;

    const isRingExt = ringTip.y < ringPip;

    const isPinkyExt = pinkyTip.y < pinkyPip;


    const distPinch = Math.hypot(thumbTip.x - indexTip.x, thumbTip.y - indexTip.y);


    // --- 握拳 (暫停) ---

    const isFist = (!isIndexExt && !isMiddleExt && !isRingExt && !isPinkyExt);

    if (isFist) {

        gestureType = "FIST";

        // 🛡️ 防誤觸檢查

        if (now < pauseProtectionTime) {

            return gestureType; 

        }


        if (now - lastGestureTime > 800 && isPlaying) { 

            audioPlayer.pause();

            isPlaying = false;

            updateStatusUI();

            lastGestureTime = now;

        }

        return gestureType;

    }


    // --- 食指 (下一首) ---

    const isIndexPoint = isIndexExt && !isMiddleExt && !isRingExt && !isPinkyExt;

    if (isIndexPoint) {

        gestureType = "INDEX";

        if (now - lastGestureTime > 1500 && isPlaying) {

            playNextSong();

            pauseProtectionTime = now + 1000; // 切歌後也保護 1 秒

            lastGestureTime = now;

        }

        return gestureType;

    }


    // --- OK (播放) ---

    if (distPinch < 0.05 && isMiddleExt) {

        gestureType = "OK";

        if (now - lastGestureTime > 1000) {

            if (!isPlaying) {

                if (!audioPlayer.src) playNextSong(); 

                else audioPlayer.play();

                

                isPlaying = true; 

                updateStatusUI();

                pauseProtectionTime = now + 2000; // 🛡️ 啟動播放時，保護 2 秒

            }

            lastGestureTime = now;

        }

        return gestureType;

    }


    // --- 音量 (手掌高度) ---

    if (isPlaying && !isFist && !isIndexPoint && distPinch > 0.1) {

        const palmY = lm[9].y;

        let targetVol = 1 - palmY; 

        targetVol = (targetVol - 0.2) * 1.6;

        targetVol = Math.max(0, Math.min(1, targetVol));

        smoothedVolume += (targetVol - smoothedVolume) * 0.1;

        

        if (Math.abs(currentVolume - smoothedVolume) > 0.01) {

            currentVolume = smoothedVolume;

            audioPlayer.volume = currentVolume;

            document.getElementById("vol-display").innerText = Math.round(currentVolume * 100) + "%";

        }

    }

    

    return gestureType;

}


function updateStatusUI() {

    const playStatus = document.getElementById("play-status");

    const boxStatus = document.getElementById("box-status");

    if (isPlaying) {

        playStatus.innerText = "Playing ▶";

        playStatus.style.color = "#00ff00";

        boxStatus.style.borderColor = "#00ff00";

    } else {

        playStatus.innerText = "Paused ⏸";

        playStatus.style.color = "#ff5555";

        boxStatus.style.borderColor = "#ff5555";

    }

}


// ==========================================

// 6. 繪圖

// ==========================================

function drawHand(landmarks, gestureType) {

    canvasCtx.lineWidth = 2;

    let mainColor = "rgba(0, 255, 204, 0.6)"; 

    let tipColor = "#00ffcc";

    

    if (gestureType === "OK") { mainColor = "rgba(0, 255, 0, 0.8)"; tipColor = "#00ff00"; }

    if (gestureType === "FIST") { mainColor = "rgba(255, 85, 85, 0.8)"; tipColor = "#ff5555"; }

    if (gestureType === "INDEX") { mainColor = "rgba(255, 204, 0, 0.8)"; tipColor = "#ffcc00"; }


    const connections = [[0,1],[1,2],[2,3],[3,4], [0,5],[5,6],[6,7],[7,8], [5,9],[9,10],[10,11],[11,12], [9,13],[13,14],[14,15],[15,16], [13,17],[17,18],[18,19],[19,20], [0,17]];

    

    canvasCtx.strokeStyle = mainColor;

    for (const [start, end] of connections) {

        const p1 = landmarks[start];

        const p2 = landmarks[end];

        canvasCtx.beginPath();

        canvasCtx.moveTo(p1.x * canvasElement.width, p1.y * canvasElement.height);

        canvasCtx.lineTo(p2.x * canvasElement.width, p2.y * canvasElement.height);

        canvasCtx.stroke();

    }


    for (let i = 0; i < landmarks.length; i++) {

        const p = landmarks[i];

        canvasCtx.beginPath();

        canvasCtx.arc(p.x * canvasElement.width, p.y * canvasElement.height, i % 4 === 0 ? 5 : 3, 0, 2 * Math.PI);

        canvasCtx.fillStyle = (i === 4 || i === 8) ? tipColor : mainColor;

        canvasCtx.fill();

    }

}
